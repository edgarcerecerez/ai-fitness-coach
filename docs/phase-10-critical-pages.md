2. **Create Recommendations List Component**
   ```typescript
   // src/components/recommendations/recommendations-list.tsx
   'use client'

   import { useEffect, useState } from 'react'
   import { createBrowserClient } from '@/utils/supabase/client'
   import { Card, CardHeader, CardTitle, CardDescription, CardContent, CardFooter } from '@/components/ui/card'
   import { Badge } from '@/components/ui/badge'
   import { Button } from '@/components/ui/button'
   import {
     TrendingUp,
     Activity,
     Apple,
     Moon,
     CheckCircle,
     X
   } from 'lucide-react'
   import { formatDistanceToNow } from 'date-fns'

   interface Recommendation {
     id: string
     user_id: string
     recommendation_type: 'nutrition' | 'exercise' | 'sleep' | 'weight'
     title: string
     description: string
     priority: 'low' | 'medium' | 'high'
     status: 'active' | 'completed' | 'dismissed'
     data_sources: string[]
     confidence_score: number
     created_at: string
     expires_at: string | null
   }

   const typeIcons = {
     nutrition: Apple,
     exercise: Activity,
     sleep: Moon,
     weight: TrendingUp,
   }

   const priorityColors = {
     low: 'bg-secondary text-secondary-foreground',
     medium: 'bg-accent text-accent-foreground',
     high: 'bg-primary text-primary-foreground',
   }

   export function RecommendationsList() {
     const [recommendations, setRecommendations] = useState<Recommendation[]>([])
     const [loading, setLoading] = useState(true)
     const [filter, setFilter] = useState<'all' | 'active' | 'completed'>('active')

     useEffect(() => {
       fetchRecommendations()
     }, [filter])

     async function fetchRecommendations() {
       setLoading(true)
       const supabase = createBrowserClient()
       const { data: { user } } = await supabase.auth.getUser()

       if (!user) return

       let query = supabase
         .from('ai_recommendations')
         .select('*')
         .eq('user_id', user.id)
         .order('created_at', { ascending: false })

       if (filter !== 'all') {
         query = query.eq('status', filter)
       }

       const { data, error } = await query

       if (!error && data) {
         setRecommendations(data)
       }

       setLoading(false)
     }

     async function markCompleted(id: string) {
       const supabase = createBrowserClient()
       await supabase
         .from('ai_recommendations')
         .update({ status: 'completed' })
         .eq('id', id)

       fetchRecommendations()
     }

     async function dismiss(id: string) {
       const supabase = createBrowserClient()
       await supabase
         .from('ai_recommendations')
         .update({ status: 'dismissed' })
         .eq('id', id)

       fetchRecommendations()
     }

     if (loading) {
       return <div>Loading recommendations...</div>
     }

     if (recommendations.length === 0) {
       return (
         <Card>
           <CardContent className="py-12 text-center">
             <p className="text-muted-foreground">
               No recommendations yet. Keep logging your health data to get personalized insights!
             </p>
           </CardContent>
         </Card>
       )
     }

     return (
       <div className="space-y-4">
         {recommendations.map((rec) => {
           const Icon = typeIcons[rec.recommendation_type]
           return (
             <Card key={rec.id}>
               <CardHeader>
                 <div className="flex items-start justify-between">
                   <div className="flex items-start gap-3">
                     <div className="mt-1">
                       <Icon className="w-5 h-5 text-primary" />
                     </div>
                     <div>
                       <CardTitle className="text-lg">{rec.title}</CardTitle>
                       <CardDescription className="mt-1">
                         {rec.description}
                       </CardDescription>
                     </div>
                   </div>
                   <Badge className={priorityColors[rec.priority]}>
                     {rec.priority}
                   </Badge>
                 </div>
               </CardHeader>

               <CardContent>
                 <div className="flex items-center gap-4 text-sm text-muted-foreground">
                   <span>
                     {formatDistanceToNow(new Date(rec.created_at), { addSuffix: true })}
                   </span>
                   <span>•</span>
                   <span>Confidence: {Math.round(rec.confidence_score * 100)}%</span>
                   <span>•</span>
                   <span>Based on: {rec.data_sources.join(', ')}</span>
                 </div>
               </CardContent>

               {rec.status === 'active' && (
                 <CardFooter className="gap-2">
                   <Button
                     size="sm"
                     onClick={() => markCompleted(rec.id)}
                   >
                     <CheckCircle className="w-4 h-4 mr-2" />
                     Mark as Done
                   </Button>
                   <Button
                     size="sm"
                     variant="ghost"
                     onClick={() => dismiss(rec.id)}
                   >
                     <X className="w-4 h-4 mr-2" />
                     Dismiss
                   </Button>
                 </CardFooter>
               )}
             </Card>
           )
         })}
       </div>
     )
   }
   ```

   ```typescript
   // src/components/mood-sleep/mood-logger.tsx
   'use client'

   import { useState } from 'react'
   import { createBrowserClient } from '@/utils/supabase/client'
   import { Card, CardHeader, CardTitle, CardContent } from '@/components/ui/card'
   import { Button } from '@/components/ui/button'
   import { Textarea } from '@/components/ui/textarea'
   import { Label } from '@/components/ui/label'
   import { toast } from 'sonner'
   import {
     Frown,
     Meh,
     Smile,
     Laugh,
     Heart
   } from 'lucide-react'

   const moodOptions = [
     { value: 1, label: 'Very Bad', icon: Frown, color: 'text-destructive' },
     { value: 2, label: 'Bad', icon: Meh, color: 'text-muted-foreground' },
     { value: 3, label: 'Okay', icon: Smile, color: 'text-accent' },
     { value: 4, label: 'Good', icon: Laugh, color: 'text-primary' },
     { value: 5, label: 'Excellent', icon: Heart, color: 'text-primary' },
   ]

   export function MoodLogger() {
     const [selectedMood, setSelectedMood] = useState<number | null>(null)
     const [notes, setNotes] = useState('')
     const [loading, setLoading] = useState(false)

     async function handleSubmit() {
       if (selectedMood === null) {
         toast.error('Please select a mood')
         return
       }

       setLoading(true)
       const supabase = createBrowserClient()
       const { data: { user } } = await supabase.auth.getUser()

       if (!user) {
         toast.error('Not authenticated')
         setLoading(false)
         return
       }

       const { error } = await supabase.from('mood_logs').insert({
         user_id: user.id,
         mood_score: selectedMood,
         notes: notes || null,
         logged_at: new Date().toISOString(),
       })

       setLoading(false)

       if (error) {
         toast.error('Failed to log mood')
       } else {
         toast.success('Mood logged successfully')
         setSelectedMood(null)
         setNotes('')
       }
     }

     return (
       <Card>
         <CardHeader>
           <CardTitle>How are you feeling today?</CardTitle>
         </CardHeader>
         <CardContent className="space-y-6">
           <div className="grid grid-cols-5 gap-2">
             {moodOptions.map((mood) => {
               const Icon = mood.icon
               const isSelected = selectedMood === mood.value
               return (
                 <button
                   key={mood.value}
                   onClick={() => setSelectedMood(mood.value)}
                   className={`
                     flex flex-col items-center gap-2 p-4 rounded-lg border-2 transition-all
                     ${isSelected
                       ? 'border-primary bg-primary/10'
                       : 'border-border hover:border-primary/50'
                     }
                   `}
                 >
                   <Icon className={`w-8 h-8 ${mood.color}`} />
                   <span className="text-xs font-medium text-foreground">
                     {mood.label}
                   </span>
                 </button>
               )
             })}
           </div>

           <div className="space-y-2">
             <Label htmlFor="mood-notes">Notes (optional)</Label>
             <Textarea
               id="mood-notes"
               placeholder="What's contributing to your mood today?"
               value={notes}
               onChange={(e) => setNotes(e.target.value)}
               rows={3}
             />
           </div>

           <Button
             onClick={handleSubmit}
             disabled={loading || selectedMood === null}
             className="w-full"
           >
             {loading ? 'Logging...' : 'Log Mood'}
           </Button>
         </CardContent>
       </Card>
     )
   }
   ```

3. **Create Sleep Logger Component**
   ```typescript
   // src/components/mood-sleep/sleep-logger.tsx
   'use client'

   import { useState } from 'react'
   import { createBrowserClient } from '@/utils/supabase/client'
   import { Card, CardHeader, CardTitle, CardContent } from '@/components/ui/card'
   import { Button } from '@/components/ui/button'
   import { Input } from '@/components/ui/input'
   import { Label } from '@/components/ui/label'
   import { Textarea } from '@/components/ui/textarea'
   import { toast } from 'sonner'

   export function SleepLogger() {
     const [hours, setHours] = useState('')
     const [quality, setQuality] = useState<number>(3)
     const [notes, setNotes] = useState('')
     const [loading, setLoading] = useState(false)

     async function handleSubmit() {
       if (!hours || parseFloat(hours) <= 0) {
         toast.error('Please enter valid sleep hours')
         return
       }

       setLoading(true)
       const supabase = createBrowserClient()
       const { data: { user } } = await supabase.auth.getUser()

       if (!user) {
         toast.error('Not authenticated')
         setLoading(false)
         return
       }

       const { error } = await supabase.from('mood_logs').insert({
         user_id: user.id,
         sleep_hours: parseFloat(hours),
         sleep_quality: quality,
         notes: notes || null,
         logged_at: new Date().toISOString(),
       })

       setLoading(false)

       if (error) {
         toast.error('Failed to log sleep')
       } else {
         toast.success('Sleep logged successfully')
         setHours('')
         setQuality(3)
         setNotes('')
       }
     }

     return (
       <Card>
         <CardHeader>
           <CardTitle>How did you sleep?</CardTitle>
         </CardHeader>
         <CardContent className="space-y-4">
           <div className="space-y-2">
             <Label htmlFor="sleep-hours">Hours of Sleep</Label>
             <Input
               id="sleep-hours"
               type="number"
               step="0.5"
               min="0"
               max="24"
               placeholder="8.0"
               value={hours}
               onChange={(e) => setHours(e.target.value)}
             />
           </div>

           <div className="space-y-2">
             <Label>Sleep Quality (1-5)</Label>
             <div className="flex gap-2">
               {[1, 2, 3, 4, 5].map((val) => (
                 <button
                   key={val}
                   onClick={() => setQuality(val)}
                   className={`
                     flex-1 py-2 rounded border-2 transition-all
                     ${quality === val
                       ? 'border-primary bg-primary/10 text-primary'
                       : 'border-border hover:border-primary/50 text-muted-foreground'
                     }
                   `}
                 >
                   {val}
                 </button>
               ))}
             </div>
           </div>

           <div className="space-y-2">
             <Label htmlFor="sleep-notes">Notes (optional)</Label>
             <Textarea
               id="sleep-notes"
               placeholder="Any dreams, disturbances, or observations?"
               value={notes}
               onChange={(e) => setNotes(e.target.value)}
               rows={3}
             />
           </div>

           <Button
             onClick={handleSubmit}
             disabled={loading || !hours}
             className="w-full"
           >
             {loading ? 'Logging...' : 'Log Sleep'}
           </Button>
         </CardContent>
       </Card>
     )
   }
   ```

2. **Create Weight Logger Component**
   ```typescript
   // src/components/weight/weight-logger.tsx
   'use client'

   import { useState } from 'react'
   import { createBrowserClient } from '@/utils/supabase/client'
   import { Card, CardHeader, CardTitle, CardContent } from '@/components/ui/card'
   import { Button } from '@/components/ui/button'
   import { Input } from '@/components/ui/input'
   import { Label } from '@/components/ui/label'
   import { toast } from 'sonner'

   export function WeightLogger() {
     const [weight, setWeight] = useState('')
     const [unit, setUnit] = useState<'kg' | 'lbs'>('kg')
     const [loading, setLoading] = useState(false)

     async function handleSubmit() {
       const weightValue = parseFloat(weight)
       if (!weight || weightValue <= 0) {
         toast.error('Please enter a valid weight')
         return
       }

       setLoading(true)
       const supabase = createBrowserClient()
       const { data: { user } } = await supabase.auth.getUser()

       if (!user) {
         toast.error('Not authenticated')
         setLoading(false)
         return
       }

       // Convert to kg if needed
       const weightKg = unit === 'lbs' ? weightValue * 0.453592 : weightValue

       const { error } = await supabase.from('weight_logs').insert({
         user_id: user.id,
         weight_kg: weightKg,
         source: 'manual',
         measured_at: new Date().toISOString(),
       })

       setLoading(false)

       if (error) {
         toast.error('Failed to log weight')
       } else {
         toast.success('Weight logged successfully')
         setWeight('')
       }
     }

     return (
       <Card>
         <CardHeader>
           <CardTitle>Log Weight</CardTitle>
         </CardHeader>
         <CardContent className="space-y-4">
           <div className="flex gap-4">
             <div className="flex-1 space-y-2">
               <Label htmlFor="weight">Weight</Label>
               <Input
                 id="weight"
                 type="number"
                 step="0.1"
                 min="0"
                 placeholder="75.5"
                 value={weight}
                 onChange={(e) => setWeight(e.target.value)}
               />
             </div>

             <div className="space-y-2">
               <Label>Unit</Label>
               <div className="flex gap-2">
                 <Button
                   variant={unit === 'kg' ? 'default' : 'outline'}
                   onClick={() => setUnit('kg')}
                   className="w-16"
                 >
                   kg
                 </Button>
                 <Button
                   variant={unit === 'lbs' ? 'default' : 'outline'}
                   onClick={() => setUnit('lbs')}
                   className="w-16"
                 >
                   lbs
                 </Button>
               </div>
             </div>
           </div>

           <Button
             onClick={handleSubmit}
             disabled={loading || !weight}
             className="w-full"
           >
             {loading ? 'Logging...' : 'Log Weight'}
           </Button>
         </CardContent>
       </Card>
     )
   }
   ```
